import java.util.Scanner;

class SecondFrequentCharacter {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter a string: ");
        String str = sc.nextLine();

        int[] count = new int[256];

        for (int i = 0; i < str.length(); i++) {
            count[str.charAt(i)]++;
        }

        int first = 0;
        int second = 0;

        char firstChar = ' ';
        char secondChar = ' ';

        for (int i = 0; i < str.length(); i++) {
            char ch = str.charAt(i);

            if (count[ch] > first) {
                second = first;
                secondChar = firstChar;

                first = count[ch];
                firstChar = ch;
            } else if (count[ch] > second && count[ch] < first) {
                second = count[ch];
                secondChar = ch;
            }
        }

        if (second == 0) {
            System.out.println("There is no second most frequent character");
        } else {
            System.out.println("Second most frequent character = " + secondChar);
            System.out.println("Frequency = " + second);
        }

        sc.close();
    }
}
